# Problem Name - Friendly Numbers  - 100B

# Description

I first parsed the comma-separated input array and safely filtered out any zeros to prevent division-by-zero errors. I then sorted the remaining numbers by their absolute values, which naturally organizes the sequence from the smallest potential divisor to the largest potential multiple. Next, i iterated through this sorted array to verify that each number is perfectly divisible by the one immediately preceding it. If the entire sequence satisfies this continuous divisibility chain, the numbers are output as "FRIENDS" otherwise "NOT FRIENDS"

# Code

```pike
int main() {
    string line1 = Stdio.stdin->gets();
    if (!line1) return 0;
    
    string line2 = Stdio.stdin->gets();
    while (line2 != 0 && String.trim_all_whites(line2) == "") {
        line2 = Stdio.stdin->gets();
    }
    if (!line2) return 0;

    array(string) arr = line2 / ",";
    int length = sizeof(arr);
    
    array(int) non_zeros = allocate(length);
    int nz_length = 0;
    
    for(int i = 0; i < length; i++) {
        int val = (int)arr[i];
        if (val != 0) {
            non_zeros[nz_length++] = val;
        }
    }
    
    if (nz_length > 1) {
        non_zeros = non_zeros[0..nz_length-1];
        array(int) keys = allocate(nz_length);
        
        for(int i = 0; i < nz_length; i++) {
            keys[i] = non_zeros[i] < 0 ? -non_zeros[i] : non_zeros[i];
        }
        
        sort(keys, non_zeros);
        
        for(int i = 1; i < nz_length; i++) {
            if (non_zeros[i] % non_zeros[i-1] != 0) {
                write("NOT FRIENDS\n");
                return 0;
            }
        }
    }
    
    write("FRIENDS\n");
    return 0;
}
```
