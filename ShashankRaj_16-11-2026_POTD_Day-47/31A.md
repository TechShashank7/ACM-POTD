# Problem Name - Worms Evolution - 31A

# Description

I sraighforwardly checked every possible 3-numbered combinations and checked if it satisfies our required condition. If yes then i print it and exit the program from there
# Code

```java

import java.util.*;
public class WormsEvolution {
    public static void main(String[] args){
        Scanner sc=new Scanner(System.in);
        int n=sc.nextInt();
        int arr[]=new int[n];
        for(int i=0;i<n;i++){
            arr[i]=sc.nextInt();
        }
        for(int i=0;i<n-2;i++){
            for(int j=i+1;j<n-1;j++){
                for(int k=j+1;k<n;k++){
                    if(arr[i]+arr[j]==arr[k]){
                        System.out.println((k+1)+" "+(i+1)+" "+(j+1));
                        return;
                    }
                    if(arr[i]+arr[k]==arr[j]){
                        System.out.println((j+1)+" "+(i+1)+" "+(k+1));
                        return;
                    }
                    if(arr[j]+arr[k]==arr[i]){
                        System.out.println((i+1)+" "+(j+1)+" "+(k+1));
                        return;
                    }
                }
            }
        }
        System.out.println(-1);
    }
}

```
